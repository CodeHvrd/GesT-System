/**
 * Gestión del local — API para cargar gastos desde la app a esta planilla.
 *
 * Cómo instalarlo (una sola vez):
 *   1. Abrí la planilla de Google → menú Extensiones → Apps Script.
 *   2. Borrá lo que haya y pegá todo este archivo.
 *   3. Cambiá el TOKEN por uno tuyo (cualquier texto largo y secreto).
 *   4. Guardá (Ctrl+S) y ejecutá la función "probar" para autorizar permisos.
 *   5. Implementar → Nueva implementación → Tipo: Aplicación web.
 *        · Ejecutar como: Yo
 *        · Quién tiene acceso: Cualquier persona
 *      Implementar, autorizar y copiar la URL que termina en /exec.
 *   6. En la app → Enlaces → Envío automático: pegá la URL y el mismo TOKEN.
 *
 * La app agrega filas al final de la hoja "Gastos" con el orden de columnas:
 *   Fecha | Mes | Tipo | Categoría | Subcategoría | Descripción | Monto | Validación
 */

var TOKEN = 'cambiar-este-token-por-uno-secreto';
var HOJA = 'Gastos';

function doGet() {
  return json({ ok: true, servicio: 'gestion-local' });
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(20000);
    var body = JSON.parse((e && e.postData && e.postData.contents) || '{}');
    if (String(body.token || '') !== TOKEN) {
      return json({ ok: false, error: 'Token inválido' });
    }
    var rows = (body.rows || []).filter(function (r) { return r && r.descripcion; });
    if (!rows.length) return json({ ok: true, agregadas: 0 });

    var sh = SpreadsheetApp.getActive().getSheetByName(HOJA);
    if (!sh) return json({ ok: false, error: 'No existe la hoja "' + HOJA + '"' });

    var start = sh.getLastRow() + 1;
    var values = rows.map(function (r) {
      return [
        r.fecha || '',
        r.mes || '',
        r.tipo || 'Variable',
        r.categoria || '',
        r.subcategoria || '',
        r.descripcion || '',
        Number(r.monto) || 0
      ];
    });

    sh.getRange(start, 1, values.length, 1).setNumberFormat('dd/MM/yyyy');
    sh.getRange(start, 2, values.length, 1).setNumberFormat('@');
    sh.getRange(start, 7, values.length, 1).setNumberFormat('$#,##0');
    sh.getRange(start, 1, values.length, 7).setValues(values);

    return json({ ok: true, agregadas: values.length });
  } catch (err) {
    return json({ ok: false, error: String(err) });
  } finally {
    try { lock.releaseLock(); } catch (ignore) {}
  }
}

function json(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}

/** Ejecutá esta función una vez para autorizar los permisos. */
function probar() {
  var sh = SpreadsheetApp.getActive().getSheetByName(HOJA);
  Logger.log(sh ? 'Hoja encontrada: ' + HOJA + ' (última fila ' + sh.getLastRow() + ')' : 'No se encontró la hoja ' + HOJA);
}
